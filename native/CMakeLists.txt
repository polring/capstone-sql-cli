cmake_minimum_required(VERSION 3.16)
project(capstone_sql LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
if(MSVC)
    add_compile_options(/W3 /utf-8)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
    add_compile_options(-Wall -Wextra)
endif()

add_library(sql_engine SHARED sql_engine/sql_engine.c)
target_compile_definitions(sql_engine PRIVATE SQL_BUILD_SHARED INTERFACE SQL_USE_SHARED)
if(UNIX)
    target_link_libraries(sql_engine PRIVATE m)
endif()
install(TARGETS sql_engine RUNTIME DESTINATION . LIBRARY DESTINATION .)

# Stage only the library built for this target for consumption by MSBuild.
if(WIN32)
    set(platform win)
    set(processor "${CMAKE_C_COMPILER_ARCHITECTURE_ID}")
elseif(APPLE)
    set(platform osx)
    set(processor "${CMAKE_OSX_ARCHITECTURES}")
else()
    set(platform linux)
endif()
if(NOT processor)
    set(processor "${CMAKE_SYSTEM_PROCESSOR}")
endif()
string(TOLOWER "${processor}" processor)
if(processor MATCHES "^(amd64|x86_64|x64)$")
    set(arch x64)
elseif(processor MATCHES "^(aarch64|arm64)$")
    set(arch arm64)
elseif(processor MATCHES "^(i[3-6]86|x86)$")
    set(arch x86)
else()
    message(FATAL_ERROR "Unsupported target architecture: ${processor}")
endif()
set(runtime_dir "${CMAKE_CURRENT_SOURCE_DIR}/../app/runtimes/${platform}-${arch}/native")
add_custom_command(TARGET sql_engine POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${runtime_dir}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:sql_engine>" "${runtime_dir}/"
    VERBATIM)

include(CTest)
if(BUILD_TESTING)
    set(fixtures "${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/v2")
    add_executable(error_test sql_engine/tests/error_test.c)
    target_link_libraries(error_test PRIVATE sql_engine)
    add_test(NAME rejection COMMAND error_test "${fixtures}/base/model.gguf" "${CMAKE_CURRENT_BINARY_DIR}")

    add_executable(fixture_test sql_engine/tests/fixture_test.c)
    target_link_libraries(fixture_test PRIVATE sql_engine)
    add_executable(cache_test sql_engine/tests/cache_test.c)
    if(UNIX)
        target_link_libraries(fixture_test PRIVATE m)
        target_link_libraries(cache_test PRIVATE m)
    endif()
    foreach(suite base custom_layout no_seeds many_seeds)
        add_test(NAME fixture_${suite} COMMAND fixture_test
            "${fixtures}/${suite}/model.gguf" "${fixtures}/${suite}/cases.txt")
        add_test(NAME cache_${suite} COMMAND cache_test "${fixtures}/${suite}/model.gguf")
    endforeach()
endif()
